Q1. User-defined module to return the sum of digits and display the result in a browser
Create two files.

File 1: sum.js
function sumDigits(n) {
    n = Math.abs(Number(n));
    let sum = 0;

    while (n > 0) {
        sum += n % 10;
        n = Math.floor(n / 10);
    }

    return sum;
}

module.exports = sumDigits;


File 2: app.js
const http = require('http');
const sumDigits = require('./sum');

http.createServer((req, res) => {
    const url = new URL(req.url, 'http://localhost:3000');
    const n = Number(url.searchParams.get('n') || 1234);

    res.writeHead(200, {
        'Content-Type': 'text/html; charset=utf-8'
    });

    res.end(`<h2>Sum of digits of ${n} = ${sumDigits(n)}</h2>`);
}).listen(3000, () => {
    console.log('http://localhost:3000/?n=1234');
});


Run:
node app.js

Q11. Check whether a given name is a file or directory; display directory contents in a browser
Create checkPathServer.js.
const http = require('http');
const fs = require('fs');
const path = require('path');

http.createServer((req, res) => {
    const url = new URL(req.url, 'http://localhost:3000');
    const name = url.searchParams.get('name') || '.';
    const target = path.resolve(__dirname, name);

    // Keep the requested path inside this project directory
    if (target !== __dirname &&
        !target.startsWith(__dirname + path.sep)) {
        res.writeHead(400);
        return res.end('Invalid path');
    }

    fs.stat(target, (err, stats) => {
        res.writeHead(200, {
            'Content-Type': 'text/html; charset=utf-8'
        });

        if (err) {
            return res.end('<h2>File or directory not found</h2>');
        }

        if (stats.isFile()) {
            return res.end('<h2>It is a file</h2>');
        }

        if (stats.isDirectory()) {
            fs.readdir(target, (err, files) => {
                if (err) return res.end('Unable to read directory');

                res.end(`
                    <h2>Directory Contents</h2>
                    <ul>
                        ${files.map(f => `<li>${f}</li>`).join('')}
                    </ul>
                `);
            });
        } else {
            res.end('Unknown type');
        }
    });
}).listen(3000, () => {
    console.log('http://localhost:3000');
});


Run:
node checkPathServer.js


Open http://localhost:3000/?name=. to list the current project directory.

Q24. Build a REST API in Node.js + Express for CRUD operations on the students table

Create the table in MySQL:
CREATE DATABASE college_db;
USE college_db;

CREATE TABLE students (

Create studentApi.js.
const express = require('express');
const mysql = require('mysql2');

const app = express();
app.use(express.json());

const db = mysql.createConnection({
    host: 'localhost',
    user: 'root',
    password: 'YOUR_PASSWORD',
    database: 'college_db'
});

// CREATE: Insert student
app.post('/students', (req, res) => {
    const { id, name, course, marks } = req.body;

    db.execute(
        'INSERT INTO students VALUES (?, ?, ?, ?)',
        [id, name, course, marks],
        (err, result) => {
            if (err) return res.status(500).json({ error: err.message });

            res.status(201).json({ message: 'Student inserted' });
        }
    );
});

// READ: Fetch all students
app.get('/students', (req, res) => {
    db.query('SELECT * FROM students', (err, rows) => {
        if (err) return res.status(500).json({ error: err.message });

        res.json(rows);
    });
});

// UPDATE: Update marks
app.put('/students/:id', (req, res) => {
    const { marks } = req.body;
    const id = Number(req.params.id);

    db.execute(
        'UPDATE students SET marks = ? WHERE id = ?',
        [marks, id],
        (err, result) => {
            if (err) return res.status(500).json({ error: err.message });

            if (result.affectedRows === 0) {
                return res.status(404).json({ message: 'Student not found' });
            }

            res.json({ message: 'Student updated' });
        }
    );
});

// DELETE: Delete student
app.delete('/students/:id', (req, res) => {
    const id = Number(req.params.id);

    db.execute(
        'DELETE FROM students WHERE id = ?',
        [id],
        (err, result) => {
            if (err) return res.status(500).json({ error: err.message });

            if (result.affectedRows === 0) {
                return res.status(404).json({ message: 'Student not found' });
            }

            res.json({ message: 'Student deleted' });
        }
    );
});

app.listen(3000, () => {
    console.log('Student API running on port 3000');
});

Q25. Build a REST API using Node.js + Express + MongoDB (Mongoose) for CRUD operations on books
Install the packages:
npm install express mongoose


Create bookApi.js.
const express = require('express');
const mongoose = require('mongoose');

const app = express();
app.use(express.json());

// Connect to MongoDB
mongoose.connect('mongodb://127.0.0.1:27017/library')
    .then(() => console.log('MongoDB connected'))
    .catch(err => console.log(err.message));

// Define book schema
const bookSchema = new mongoose.Schema({
    title: String,
    author: String,
    year: Number,
    price: Number
});

const Book = mongoose.model('Book', bookSchema);

// CREATE: Insert a book
app.post('/books', async (req, res) => {
    try {
        const book = await Book.create(req.body);
        res.status(201).json(book);
    } catch (err) {
        res.status(400).json({ error: err.message });
    }
});

// READ: Fetch all books
app.get('/books', async (req, res) => {
    try {
        const books = await Book.find();
        res.json(books);
    } catch (err) {
        res.status(500).json({ error: err.message });
    }
});

// UPDATE: Update a book
app.put('/books/:id', async (req, res) => {
    try {
        const book = await Book.findByIdAndUpdate(
            req.params.id,
            req.body,
            { new: true, runValidators: true }
        );

        if (!book) {
            return res.status(404).json({ message: 'Book not found' });
        }

        res.json(book);
    } catch (err) {
        res.status(400).json({ error: err.message });
    }
});

// DELETE: Delete a book
app.delete('/books/:id', async (req, res) => {
    try {
        const book = await Book.findByIdAndDelete(req.params.id);

        if (!book) {
            return res.status(404).json({ message: 'Book not found' });
        }

        res.json({ message: 'Book deleted successfully' });
    } catch (err) {
        res.status(400).json({ error: err.message });
    }
});

app.listen(3000, () => {
    console.log('Book API running on port 3000');
});
