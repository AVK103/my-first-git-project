Q1. Create a user-defined module to check prime and perfect numbers (runtime input)

Files / setup: Create primePerfect.js and main.js.

// primePerfect.js
function isPrime(n) {
  if (n <= 1) return false;
  for (let i = 2; i <= Math.sqrt(n); i++) {
    if (n % i === 0) return false;
  }
  return true;
}
 
function isPerfect(n) {
  if (n <= 1) return false;
  let sum = 0;
  for (let i = 1; i <= n / 2; i++) {
    if (n % i === 0) sum += i;
  }
  return sum === n;
}
 
module.exports = { isPrime, isPerfect };
 
// main.js
const readline = require('readline');
const { isPrime, isPerfect } = require('./primePerfect');
const rl = readline.createInterface({
  input: process.stdin, output: process.stdout
});
 
rl.question('Enter an integer: ', answer => {
  const n = Number(answer);
  if (!Number.isInteger(n)) {
    console.log('Please enter an integer.');
  } else {
    console.log(n + (isPrime(n) ? ' is prime' : ' is not prime'));
    console.log(n + (isPerfect(n) ? ' is perfect' : ' is not perfect'));
  }
  rl.close();
});

Run: node main.js

Expected result: For input 6: “6 is not prime” and “6 is perfect”.

Exam note: A perfect number equals the sum of its positive proper divisors; 6 = 1 + 2 + 3.

Q4. Custom module to calculate factorial and use it in the main script

Files / setup: Create factorial.js and main.js.

// factorial.js
function factorial(n) {
  if (!Number.isInteger(n) || n < 0) {
    throw new Error('Enter a non-negative integer.');
  }
  let result = 1;
  for (let i = 2; i <= n; i++) result *= i;
  return result;
}
module.exports = factorial;
 
// main.js
const readline = require('readline');
const factorial = require('./factorial');
const rl = readline.createInterface({
  input: process.stdin, output: process.stdout
});
rl.question('Enter a non-negative integer: ', value => {
  try {
    console.log('Factorial = ' + factorial(Number(value)));
  } catch (err) {
    console.log(err.message);
  }
  rl.close();
});

Run: node main.js

Expected result: For input 5: Factorial = 120.

const fs = require('fs');
const readline = require('readline');
const rl = readline.createInterface({
  input: process.stdin, output: process.stdout
});
 
rl.question('Roll number: ', roll => {
  rl.question('Name: ', name => {
    rl.question('Course: ', course => {
      const file = 'students.json';
      fs.readFile(file, 'utf8', (err, data) => {
        if (err && err.code !== 'ENOENT') throw err;
        let students = [];
        if (!err && data.trim()) students = JSON.parse(data);
        students.push({ rollno: Number(roll), name, course });
        fs.writeFile(file, JSON.stringify(students, null, 2), err => {
          if (err) throw err;
          console.log('Student added.');
          rl.close();
        });
      });
    });
  });
});

Run: node addStudent.js

Expected result: The new student object is saved in students.json.

Exam note: Start with [] in students.json. In real applications, validate input and handle malformed JSON.

Q17. Upload an image file using Node.js

const express = require('express');
const multer = require('multer');
const path = require('path');
const fs = require('fs');
 
fs.mkdirSync('uploads', { recursive: true });
const storage = multer.diskStorage({
  destination: (req, file, cb) => cb(null, 'uploads/'),
  filename: (req, file, cb) =>
    cb(null, Date.now() + path.extname(file.originalname))
});
const upload = multer({
  storage,
  fileFilter: (req, file, cb) => {
    const allowed = ['image/jpeg', 'image/png', 'image/gif', 'image/webp'];
    cb(null, allowed.includes(file.mimetype));
  }
});
const app = express();
 
app.get('/', (req, res) => res.send(`
  <form method="POST" action="/upload" enctype="multipart/form-data">
    <input type="file" name="image" accept="image/*" required>
    <button>Upload image</button>
  </form>`));
 
app.post('/upload', upload.single('image'), (req, res) => {
  if (!req.file) return res.status(400).send('Please select a supported image.');
  res.send('Image uploaded: ' + req.file.filename);
});
app.listen(3000, () => console.log('http://localhost:3000'));

Run: node app.js; open http://localhost:3000

Expected result: Uploaded images are saved in the uploads folder.

Exam note: For production, also validate file contents and size; MIME type alone is not a security guarantee.
